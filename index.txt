function fileToBase64(file){
  return new Promise(function(resolve,reject){
    const r = new FileReader();
    r.onload = function(){ resolve(r.result.split(',')[1]); };
    r.onerror = reject;
    r.readAsDataURL(file);
  });
}

function hashStr(s){ let h=0; for(let i=0;i<s.length;i++){ h=(h*31+s.charCodeAt(i))>>>0 } return h }
function ean13(base12){
  let sum=0; for(let i=0;i<12;i++) sum += (+base12[i])*(i%2===0?1:3);
  return base12 + ((10-sum%10)%10);
}
function genBarcode(seed){
  let n = hashStr(String(seed)) % 1000000000;
  let bc = ean13('899' + String(n).padStart(9,'0'));
  let guard = 0;
  while(barang.some(b=>b.barcode===bc) && guard<60){ n=(n+7919)%1000000000; bc=ean13('899'+String(n).padStart(9,'0')); guard++ }
  return bc;
}
function genQR(kode){ return String(kode||'').trim().toUpperCase() }
function bcFormat(v){
  v = String(v||'').trim();
  if(/^\d{12}$/.test(v)) return 'EAN13';
  if(/^\d{13}$/.test(v)){
    let sum=0; for(let i=0;i<12;i++) sum += (+v[i])*(i%2===0?1:3);
    if(((10-sum%10)%10) === +v[12]) return 'EAN13';
  }
  if(/^\d{7}$/.test(v)) return 'EAN8';
  return 'CODE128';
}
function ensureCodes(){
  barang.forEach((b,i)=>{
    if(!b.tipeKode) b.tipeKode = 'BARCODE';
    if(!b.barcode) b.barcode = b.tipeKode==='QR' ? genQR(b.kode||('BRG'+i)) : genBarcode((b.kode||'X')+'_'+i);
  });
}

function pushCart(b,qty){
  if(b.kategori!=='Jasa'){
    const inCart = (cart.find(c=>c.kode===b.kode)||{jumlah:0}).jumlah;
    if(b.stok - inCart < qty){ toast('Stok '+b.nama+' tidak mencukupi, sisa '+b.stok+' unit','err'); return }
  }
  const ex = cart.find(c=>c.kode===b.kode);
  if(ex){ ex.jumlah+=qty; ex.subtotal=ex.jumlah*ex.harga }
  else cart.push({kode:b.kode,nama:b.nama,jumlah:qty,harga:b.harga,subtotal:b.harga*qty});
  renderCart();
}
function addToCart(){
  const kode = document.getElementById('kasirBarang').value;
  const qty = parseInt(document.getElementById('kasirJumlah').value)||1;
  if(!kode){ toast('Pilih barang terlebih dahulu','err'); return }
  const b = barang.find(x=>x.kode===kode);
  if(b) pushCart(b,qty);
}

function clearCart(){ if(cart.length && confirm('Kosongkan keranjang?')){ cart=[]; renderCart() } }
function renderCart(){
  const tb = document.getElementById('cartBody');
  tb.innerHTML = cart.length ? cart.map((c,i)=>
    '<tr>'+
      '<td><strong>'+esc(c.nama)+'</strong><br><span style="font-size:.7rem;color:var(--text-2);font-family:monospace">'+c.kode+'</span></td>'+
      '<td><input type="number" min="1" value="'+c.jumlah+'" style="width:64px;padding:.32rem;border:1px solid var(--border);border-radius:8px;background:var(--card);color:var(--text)" onchange="updQty('+i+',this.value)"></td>'+

function updQty(i,v){ cart[i].jumlah=Math.max(1,parseInt(v)||1); cart[i].subtotal=cart[i].jumlah*cart[i].harga; renderCart() }
function checkout(){
  if(!cart.length){ toast('Keranjang masih kosong','err'); return }
  const total = cart.reduce((s,c)=>s+c.subtotal,0);
  const id = 'TRX'+String(trxCounter++).padStart(3,'0');
  const jam = new Date().toLocaleTimeString('id-ID',{hour:'2-digit',minute:'2-digit'});
  transaksi.push({id,tanggal:TODAY,total,jam});
  let nextId = detailTransaksi.reduce((m,d)=>Math.max(m,d.id),0)+1;
  cart.forEach(c=>{
    detailTransaksi.push({id:nextId++,id_transaksi:id,kode_barang:c.kode,jumlah:c.jumlah,harga:c.harga,subtotal:c.subtotal});
    const b = barang.find(x=>x.kode===c.kode);
    if(b && b.kategori!=='Jasa') b.stok = Math.max(0, b.stok - c.jumlah);
  });
  save();
  cart = [];
  renderCart(); renderHistory(); renderKasirSelect();
  toast('Checkout berhasil. Transaksi '+id+' tersimpan, total '+rp(total));
}

let chartObj = null;
function reportData(){
  const today = transaksi.filter(t=>t.tanggal===TODAY);
  const ids = new Set(today.map(t=>t.id));
  const ds = detailTransaksi.filter(d=>ids.has(d.id_transaksi));
  const cats = ['Minuman','Makanan','ATK','Jasa'];
  const perCat = {}; cats.forEach(c=>perCat[c]=0);
  ds.forEach(d=>{ const b=barang.find(x=>x.kode===d.kode_barang); if(b&&perCat[b.kategori]!==undefined) perCat[b.kategori]+=d.subtotal });
  const total = today.reduce((s,t)=>s+t.total,0);
  return {total,jmlTrx:today.length,jmlItem:ds.reduce((s,d)=>s+d.jumlah,0),
    avg:today.length?Math.round(total/today.length):0,perCat,cats,details:ds};
}

        '<span class="rank-qty">'+r.qty+' terjual</span></li>').join('')
    : '<li style="color:var(--text-2)">Belum ada data penjualan.</li>';
  const low = barang.filter(b=>b.kategori!=='Jasa'&&b.stok<=10);
  document.getElementById('stockWarnings').innerHTML = low.length

      '<div class="alert '+(b.stok===0?'out':'')+'">'+

    : '<p style="color:var(--text-2);font-size:.9rem">Semua stok dalam kondisi aman, di atas 10 unit.</p>';

function renderPreview(){
  const wrap = document.getElementById('modalBcWrap');
  const val = document.getElementById('inputBarcode').value.trim();
  const tipe = document.getElementById('inputTipe').value;
  const lc = getComputedStyle(document.body).getPropertyValue('--bc').trim()||'#0A2540';
  wrap.innerHTML='';
  if(!val){ wrap.innerHTML='<span style="font-size:.8rem;color:#5A7A99">Nomor barcode atau QR Code belum diisi</span>'; return }
  if(tipe==='QR'){
    const img = document.createElement('img');
    img.className='qr'; wrap.appendChild(img);
    drawQRImg(img,val,116,lc);
  }else{
    const svg = document.createElementNS('http://www.w3.org/2000/svg','svg');
    wrap.appendChild(svg);
    drawOne(svg,val,{h:58,w:1.9,bg:'transparent',lc:lc,fmt:bcFormat(val)});
  }
  const tag = document.createElement('span');
  tag.className='code-tag';
  tag.textContent = (tipe==='QR'?'QR Code':'Barcode')+' - '+val;
  wrap.appendChild(tag);
}

function saveBarang(){
  const idx = parseInt(document.getElementById('editIndex').value);
  const kode = document.getElementById('inputKode').value.trim().toUpperCase();
  const nama = document.getElementById('inputNama').value.trim();
  const kategori = document.getElementById('inputKategori').value;
  const harga = parseInt(document.getElementById('inputHarga').value)||0;
  let stok = parseInt(document.getElementById('inputStok').value)||0;
  const tipeKode = document.getElementById('inputTipe').value;
  let barcode = document.getElementById('inputBarcode').value.trim();
  if(!kode||!nama){ toast('Kode dan nama wajib diisi','err'); return }
  if(harga<=0){ toast('Harga harus lebih dari nol','err'); return }
  if(barang.some((b,i)=>b.kode===kode && i!==idx)){ toast('Kode barang sudah dipakai','err'); return }
  if(!barcode) barcode = tipeKode==='QR' ? genQR(kode) : genBarcode(kode+'_'+Date.now());
  if(barang.some((b,i)=>String(b.barcode).toLowerCase()===barcode.toLowerCase() && i!==idx)){ toast('Nomor barcode atau QR sudah dipakai barang lain','err'); return }
  if(kategori==='Jasa') stok = 9999;
  if(idx>=0) barang[idx] = {kode,nama,kategori,harga,stok,barcode,tipeKode};
  else barang.push({kode,nama,kategori,harga,stok,barcode,tipeKode});
  save(); renderDataTable(); closeModal('barangModal');
  toast(idx>=0?'Data barang diperbarui':'Barang berhasil ditambahkan');
}
